```python
"""Rutas, constantes y carga de credenciales de badgegen."""
from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path

CANVAS_SIZE = 1254
TARGET_DIAMETER = 1180
ROUNDNESS_THRESHOLD = 0.985
MIN_FIXABLE_RATIO = 0.95
JSDELIVR_PURGE = "https://purge.jsdelivr.net/gh/makespacemadrid/open-badges@main/{file}"


def load_env_file(path: Path) -> None:
    """Carga KEY=VALUE de un .env en os.environ sin pisar valores existentes."""
    if not path.is_file():
        return
    for line in path.read_text(encoding="utf-8").splitlines():
        line = line.strip()
        if not line or line.startswith("#") or "=" not in line:
            continue
        key, value = line.split("=", 1)
        os.environ.setdefault(key.strip(), value.strip())


@dataclass(frozen=True)
class Settings:
    repo_root: Path

    @classmethod
    def locate(cls) -> "Settings":
        root = Path(__file__).resolve().parents[2]
        settings = cls(repo_root=root)
        load_env_file(settings.env_path)
        return settings

    @property
    def manifest_path(self) -> Path:
        return self.repo_root / "badges.yaml"

    @property
    def candidates_dir(self) -> Path:
        return self.repo_root / "candidates"

    @property
    def review_dir(self) -> Path:
        return self.repo_root / "review"

    @property
    def readme_path(self) -> Path:
        return self.repo_root / "README.md"

    @property
    def missing_path(self) -> Path:
        return self.repo_root / "missing-images.md"

    @property
    def env_path(self) -> Path:
        return self.repo_root / "tools" / ".env"

    @property
    def catalog_path(self) -> Path:
        env = os.environ.get("BADGEGEN_CATALOG")
        if env:
            return Path(env)
        return self.repo_root.parent / "InfraSelfService" / "src" / "lib" / "badges" / "catalog.ts"
```

```python
"""Carga/guardado del manifiesto badges.yaml (fuente de verdad local)."""
from __future__ import annotations

from dataclasses import dataclass, field
from pathlib import Path

import yaml

from badgegen import config

DEFAULT_STYLE = (
    "Circular maker badge emblem, flat vector style, golden yellow (#F5C400) artwork "
    "on a black disc, outer yellow ring with small screw heads, bold sans-serif lettering, "
    "high contrast, clean centered composition, no gradients, no photo textures"
)
DEFAULT_REFERENCES = ["badge-template.png", "badge-member.png", "badge-hello-glados.png"]

STATUSES = ("ok", "missing", "regen")


@dataclass
class Badge:
    slug: str
    name: str
    file: str
    track: str = ""
    description: str = ""
    criteria: str = ""
    visual: str | None = None
    status: str = "missing"
    issues: list[str] = field(default_factory=list)
    provider: str | None = None
    prompt_extra: str | None = None

    def to_dict(self) -> dict:
        d = {
            "slug": self.slug,
            "name": self.name,
            "file": self.file,
            "track": self.track,
            "description": self.description,
            "criteria": self.criteria,
            "status": self.status,
            "issues": list(self.issues),
        }
        for key in ("visual", "provider", "prompt_extra"):
            value = getattr(self, key)
            if value is not None:
                d[key] = value
        return d

    @classmethod
    def from_dict(cls, d: dict) -> "Badge":
        return cls(
            slug=d["slug"],
            name=d.get("name", d["slug"]),
            file=d.get("file", f"{d['slug']}.png"),
            track=d.get("track", ""),
            description=d.get("description", ""),
            criteria=d.get("criteria", ""),
            visual=d.get("visual"),
            status=d.get("status", "missing"),
            issues=list(d.get("issues") or []),
            provider=d.get("provider"),
            prompt_extra=d.get("prompt_extra"),
        )


@dataclass
class Manifest:
    style: str = DEFAULT_STYLE
    reference_images: list[str] = field(default_factory=lambda: list(DEFAULT_REFERENCES))
    target_diameter: int = config.TARGET_DIAMETER
    roundness_threshold: float = config.ROUNDNESS_THRESHOLD
    badges: list[Badge] = field(default_factory=list)

    def get(self, slug: str) -> Badge | None:
        for badge in self.badges:
            if badge.slug == slug:
                return badge
        return None


def load(path: Path) -> Manifest:
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    return Manifest(
        style=data.get("style", DEFAULT_STYLE),
        reference_images=list(data.get("reference_images") or DEFAULT_REFERENCES),
        target_diameter=int(data.get("target_diameter", config.TARGET_DIAMETER)),
        roundness_threshold=float(data.get("roundness_threshold", config.ROUNDNESS_THRESHOLD)),
        badges=[Badge.from_dict(b) for b in data.get("badges") or []],
    )


def save(m: Manifest, path: Path) -> None:
    data = {
        "style": m.style,
        "reference_images": list(m.reference_images),
        "target_diameter": m.target_diameter,
        "roundness_threshold": m.roundness_threshold,
        "badges": [b.to_dict() for b in m.badges],
    }
    text = yaml.safe_dump(data, sort_keys=False, allow_unicode=True, width=100)
    path.write_text(text, encoding="utf-8")
```

```python
"""Parser del catálogo TypeScript del portal (const BADGES: BadgeSpec[])."""
from __future__ import annotations

import re
from dataclasses import dataclass
from pathlib import Path


class CatalogParseError(Exception):
    pass


@dataclass
class CatalogEntry:
    slug: str
    name: str
    description: str
    criteria: str
    track: str
    image_file: str | None
    retired: bool


def default_file(slug: str) -> str:
    return f"{slug}.png" if slug.startswith("badge-") else f"badge-{slug}.png"


def _split_entries(text: str) -> list[str]:
    """Devuelve los literales {...} de primer nivel del array BADGES.

    Scanner de caracteres consciente de strings ('", `) con escapes y de
    anidamiento de llaves/corchetes: los regex no valen porque las entradas
    contienen questGroups: [{...}] anidados.
    """
    marker = re.search(r"const\s+BADGES\s*:[^=]*=\s*\[", text)
    if not marker:
        raise CatalogParseError("no se encontró 'const BADGES' en el catálogo")
    i = marker.end()
    entries: list[str] = []
    depth_braces = 0
    depth_brackets = 1  # ya estamos dentro del [
    in_string: str | None = None
    start = None
    while i < len(text):
        ch = text[i]
        if in_string:
            if ch == "\\":
                i += 2
                continue
            if ch == in_string:
                in_string = None
        elif ch in ("'", '"', "`"):
            in_string = ch
        elif ch == "{":
            if depth_braces == 0 and depth_brackets == 1:
                start = i
            depth_braces += 1
        elif ch == "}":
            depth_braces -= 1
            if depth_braces == 0 and start is not None:
                entries.append(text[start : i + 1])
                start = None
        elif ch == "[":
            depth_brackets += 1
        elif ch == "]":
            depth_brackets -= 1
            if depth_brackets == 0:
                break
        i += 1
    if not entries:
        raise CatalogParseError("0 entradas en BADGES: ¿cambió el formato del catálogo?")
    return entries


_STRING = r"'((?:[^'\\]|\\.)*)'"


def _field(chunk: str, key: str) -> str | None:
    m = re.search(rf"(?<![A-Za-z]){key}:\s*{_STRING}", chunk, re.DOTALL)
    if not m:
        return None
    return m.group(1).replace("\\'", "'").replace("\\\\", "\\")


def parse_catalog(path: Path) -> list[CatalogEntry]:
    text = path.read_text(encoding="utf-8")
```

```python
"""QC geométrico de imágenes de badge: tamaño, transparencia, redondez."""
from __future__ import annotations

from dataclasses import dataclass, field

import numpy as np
from PIL import Image

from badgegen import config


def outer_mask(im: Image.Image) -> np.ndarray:
    arr = np.asarray(im.convert("RGBA"))
    alpha = arr[:, :, 3]
    if alpha.min() < 128:  # hay transparencia real
        return alpha > 128
    rgb = arr[:, :, :3].astype(int)
    corners = np.array([rgb[0, 0], rgb[0, -1], rgb[-1, 0], rgb[-1, -1]])
    bg = np.median(corners, axis=0)
    return np.abs(rgb - bg).sum(axis=2) > 60


def mask_bbox(mask: np.ndarray, min_count: int = 3) -> tuple[int, int, int, int]:
    cols = np.nonzero(mask.sum(axis=0) >= min_count)[0]
    rows = np.nonzero(mask.sum(axis=1) >= min_count)[0]
    if len(cols) == 0 or len(rows) == 0:
        raise ValueError("máscara vacía")
    return int(cols[0]), int(rows[0]), int(cols[-1]), int(rows[-1])


def roundness(mask: np.ndarray) -> float:
    ys, xs = np.nonzero(mask)
    if len(xs) < 10:
        return 0.0
    cov = np.cov(np.stack([xs - xs.mean(), ys - ys.mean()]))
    ev = np.linalg.eigvalsh(cov)
    if ev[1] <= 0:
        return 0.0
    return float((max(ev[0], 0.0) / ev[1]) ** 0.5)


@dataclass
class QCResult:
    roundness: float = 0.0
    issues: list[str] = field(default_factory=list)

    @property
    def ok(self) -> bool:
        return not self.issues


def check_image(im: Image.Image, *, canvas: int = config.CANVAS_SIZE,
                threshold: float = config.ROUNDNESS_THRESHOLD) -> QCResult:
    result = QCResult()
    if im.size != (canvas, canvas):
        result.issues.append("wrong-size")
    rgba = im.convert("RGBA")
    if np.asarray(rgba)[:, :, 3].min() >= 128:
        result.issues.append("no-transparency")
    mask = outer_mask(rgba)
    if mask.sum() < 100:
        result.issues.append("empty-image")
        return result
    result.roundness = roundness(mask)
    if result.roundness < threshold:
        result.issues.append("flattened-circle")
    return result
```

```python
"""Normalización de badges: circularizar el disco, centrar y máscara alpha."""
from __future__ import annotations

from dataclasses import dataclass

import numpy as np
from PIL import Image, ImageDraw

from badgegen import config, qc


class ShapeError(Exception):
    pass


@dataclass
class DiskInfo:
    x0: int
    y0: int
    x1: int
    y1: int

    @property
    def w(self) -> int:
        return self.x1 - self.x0 + 1

    @property
    def h(self) -> int:
        return self.y1 - self.y0 + 1

    @property
    def cx(self) -> float:
        return (self.x0 + self.x1) / 2

    @property
    def cy(self) -> float:
        return (self.y0 + self.y1) / 2

    @property
    def ratio(self) -> float:
        return min(self.w, self.h) / max(self.w, self.h)


def analyze(im: Image.Image) -> DiskInfo:
    mask = qc.outer_mask(im)
    try:
        x0, y0, x1, y1 = qc.mask_bbox(mask)
    except ValueError as exc:
        raise ShapeError("no se detecta ningún disco en la imagen") from exc
    return DiskInfo(x0, y0, x1, y1)


def _circle_alpha(canvas: int, diameter: int) -> np.ndarray:
    """Máscara circular antialiased (supersampling 4x)."""
    scale = 4
    big = Image.new("L", (canvas * scale, canvas * scale), 0)
    d = ImageDraw.Draw(big)
    c = canvas * scale / 2
    r = (diameter / 2 + 1) * scale
    d.ellipse((c - r, c - r, c + r, c + r), fill=255)
    return np.asarray(big.resize((canvas, canvas), Image.LANCZOS))


def process(im: Image.Image, *, canvas: int = config.CANVAS_SIZE,
            diameter: int = config.TARGET_DIAMETER,
            min_ratio: float = config.MIN_FIXABLE_RATIO) -> Image.Image:
    im = im.convert("RGBA")
    info = analyze(im)
    if info.ratio < min_ratio:
        raise ShapeError(
            f"disco demasiado deformado (ratio {info.ratio:.3f} < {min_ratio}); regenerar"
        )
    # 1. Escala anisotrópica: el disco pasa a medir `diameter` en ambos ejes.
    sx, sy = diameter / info.w, diameter / info.h
    resized = im.resize(
        (max(1, round(im.width * sx)), max(1, round(im.height * sy))), Image.LANCZOS
    )
    info2 = analyze(resized)
    # 2. Centrar el disco en el lienzo.
    out = Image.new("RGBA", (canvas, canvas), (0, 0, 0, 0))
    ox = round(canvas / 2 - info2.cx)
    oy = round(canvas / 2 - info2.cy)
    out.paste(resized, (ox, oy), resized)
    # 3. Máscara circular: fuera del disco todo transparente.
    arr = np.asarray(out).copy()
    arr[:, :, 3] = np.minimum(arr[:, :, 3], _circle_alpha(canvas, diameter))
    return Image.fromarray(arr)
```

Nota sobre `test_process_opaque_background_gets_transparency`: con fondo opaco `out.paste(resized, ..., resized)` usa el alpha de `resized` que es 255 en todas partes — el paso 3 es el que recorta el círculo. El interior negro del disco queda opaco porque está dentro del círculo. Por eso el test comprueba centro opaco y esquina transparente.

```python
"""Utilidades comunes de providers: errores y reintentos con backoff."""
from __future__ import annotations

import time

import httpx


class ProviderError(Exception):
    pass


def post_with_retry(client: httpx.Client, url: str, *, json: dict,
                    headers: dict | None = None, retries: int = 3,
                    sleep=time.sleep) -> httpx.Response:
    for attempt in range(retries + 1):
        response = client.post(url, json=json, headers=headers, timeout=120)
        if response.status_code == 429 or response.status_code >= 500:
            if attempt == retries:
                raise ProviderError(
                    f"HTTP {response.status_code} tras {retries} reintentos: {response.text[:200]}"
                )
            sleep(2 ** (attempt + 1))
            continue
        if response.status_code >= 400:
            raise ProviderError(f"HTTP {response.status_code}: {response.text[:500]}")
        return response
    raise ProviderError("inalcanzable")
```

```python
"""Proveedor nano banana: API Gemini de generación de imagen, REST directo."""
from __future__ import annotations

import base64

import httpx

from badgegen.providers.base import ProviderError, post_with_retry

API_BASE = "https://generativelanguage.googleapis.com/v1beta"
DEFAULT_MODEL = "gemini-2.5-flash-image"


class GeminiProvider:
    name = "gemini"

    def __init__(self, api_key: str, model: str = DEFAULT_MODEL,
                 client: httpx.Client | None = None):
        self.api_key = api_key
        self.model = model
        self.client = client or httpx.Client()

    def generate(self, prompt: str, reference_images: list[bytes], n: int = 1) -> list[bytes]:
        parts: list[dict] = [{"text": prompt}]
        for ref in reference_images:
            parts.append({"inline_data": {
                "mime_type": "image/png",
                "data": base64.b64encode(ref).decode(),
            }})
        url = f"{API_BASE}/models/{self.model}:generateContent"
        headers = {"x-goog-api-key": self.api_key}
        images: list[bytes] = []
        for _ in range(n):
            response = post_with_retry(
                self.client, url, json={"contents": [{"parts": parts}]}, headers=headers
            )
            images.append(self._extract_image(response.json()))
        return images

    @staticmethod
    def _extract_image(data: dict) -> bytes:
        for candidate in data.get("candidates", []):
            for part in candidate.get("content", {}).get("parts", []):
                blob = part.get("inlineData") or part.get("inline_data")
                if blob and blob.get("data"):
                    return base64.b64decode(blob["data"])
        raise ProviderError(f"respuesta sin imagen: {str(data)[:300]}")
```

```python
class OpenAICompatProvider:

        self.api_key = api_key
        self.model = model
        self.size = size

                "prompt": prompt,
                "n": 1,
                "size": self.size,
                "response_format": "b64_json",
            })
```

```python
import base64 as _base64
import json as _json

import httpx as _httpx

JUDGE_PROMPT = """Eres un revisor de calidad de insignias (badges) circulares.
Evalúa SOLO la legibilidad y limpieza de esta imagen de badge:
- ¿El texto se lee sin esfuerzo? ¿Está bien escrito (sin letras deformadas o inventadas)?
- ¿La tipografía está entera (sin trazos rotos ni glifos corruptos)?
- ¿Hay ruido, artefactos o texturas sucias?
- ¿El contraste figura/fondo es alto?
Responde en JSON: {"score": <entero 1-5, 5 = perfecto>, "issues": ["problema en español", ...]}.
Si score >= 4, issues debe ser [] o muy corto."""


@dataclass
class JudgeResult:
    score: int
    issues: list[str] = field(default_factory=list)


def judge_legibility(png: bytes, api_key: str, model: str = "gemini-2.5-flash",
                     client: _httpx.Client | None = None) -> JudgeResult:
    from badgegen.providers.base import ProviderError, post_with_retry

    client = client or _httpx.Client()
    url = f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent"
    body = {
        "contents": [{"parts": [
            {"text": JUDGE_PROMPT},
            {"inline_data": {"mime_type": "image/png",
                             "data": _base64.b64encode(png).decode()}},
        ]}],
        "generationConfig": {"responseMimeType": "application/json"},
    }
    response = post_with_retry(client, url, json=body,
                               headers={"x-goog-api-key": api_key})
    try:
        text = response.json()["candidates"][0]["content"]["parts"][0]["text"]
        verdict = _json.loads(text)
        return JudgeResult(score=int(verdict["score"]),
                           issues=[str(i) for i in verdict.get("issues", [])])
    except (KeyError, IndexError, ValueError, TypeError) as exc:
        raise ProviderError(f"veredicto ilegible del juez: {exc}") from exc
```

```python
"""Orquestación de la generación de candidatos."""
from __future__ import annotations

import io
import json
from datetime import datetime, timezone
from pathlib import Path

from PIL import Image

from badgegen import qc
from badgegen.config import Settings
from badgegen.manifest import Badge, Manifest

HARD_RULES = (
    "Hard requirements: the badge is a PERFECT CIRCLE (never an ellipse or squashed disc), "
    "fully visible and centered with margin around it, on a plain solid black background with "
    "nothing else in the frame. Any text must be spelled EXACTLY as given, fully legible, "
    "with clean unbroken letterforms."
)


def build_prompt(m: Manifest, b: Badge) -> str:
    motif = b.visual or b.description
    lines = [
        m.style.strip(),
        f'Badge title text (must appear exactly): "{b.name}"',
        f"Main motif: {motif}",
    ]
    if b.prompt_extra:
        lines.append(b.prompt_extra)
    lines.append(HARD_RULES)
    return "\n".join(lines)


def select_badges(m: Manifest, slugs: list[str] | None,
                  statuses: list[str] | None) -> list[Badge]:
    if slugs:
        result = []
        for slug in slugs:
            badge = m.get(slug)
            if badge is None:
                raise KeyError(f"slug desconocido: {slug}")
            result.append(badge)
        return result
    if statuses:
        return [b for b in m.badges if b.status in statuses]
    return []


def _next_index(directory: Path) -> int:
    existing = [int(p.name.split("-")[0]) for p in directory.glob("*.png")
                if p.name.split("-")[0].isdigit()]
    return max(existing, default=0) + 1


def _load_references(settings: Settings, m: Manifest) -> list[bytes]:
    refs = []
    for name in m.reference_images:
        path = settings.repo_root / name
        if path.is_file():
            refs.append(path.read_bytes())
    return refs


def run_generate(settings: Settings, m: Manifest, provider,
                 slugs: list[str] | None = None, statuses: list[str] | None = None,
                 n: int = 4) -> dict:
    references = _load_references(settings, m)
    report: dict = {}
    for badge in select_badges(m, slugs, statuses):
        try:
            report[badge.slug] = _generate_one(settings, m, provider, badge, references, n)
        except Exception as exc:  # el batch sigue; el error queda en el informe
            report[badge.slug] = f"ERROR: {exc}"
    return report


def _generate_one(settings: Settings, m: Manifest, provider, badge: Badge,
                  references: list[bytes], n: int) -> list[str]:
    prompt = build_prompt(m, badge)
    directory = settings.candidates_dir / badge.slug
    directory.mkdir(parents=True, exist_ok=True)
    saved = []
    for image_bytes in provider.generate(prompt, references, n):
        index = _next_index(directory)
        stem = f"{index:02d}-{provider.name}"
        png_path = directory / f"{stem}.png"
        png_path.write_bytes(image_bytes)
        result = qc.check_image(Image.open(io.BytesIO(image_bytes)),
                                threshold=m.roundness_threshold)
        sidecar = {
            "slug": badge.slug,
            "prompt": prompt,
            "provider": provider.name,
            "model": provider.model,
            "created": datetime.now(timezone.utc).isoformat(timespec="seconds"),
            "qc": {"roundness": round(result.roundness, 4), "issues": result.issues},
        }
        (directory / f"{stem}.json").write_text(
            json.dumps(sidecar, ensure_ascii=False, indent=2), encoding="utf-8"
        )
        saved.append(str(png_path.relative_to(settings.repo_root)))
    return saved
```

Nota: el QC de candidatos usa el umbral del manifiesto pero NO el tamaño del lienzo (los candidatos vienen en la resolución nativa del modelo, p.ej. 1024×1024 — `wrong-size` en el sidecar es informativo, el post-proceso de promote lo arregla).

```python
"""Regeneración de las tablas de README.md y missing-images.md desde el manifiesto."""
from __future__ import annotations

import re

from badgegen.config import Settings
from badgegen.manifest import Badge, Manifest

START = "<!-- badgegen:tables:start -->"
END = "<!-- badgegen:tables:end -->"

TRACKS = [
    ("onboarding", "Onboarding"),
    ("maquinas", "Máquinas"),
    ("plataformas", "Plataformas"),
    ("comunidad", "Comunidad"),
    ("eventos", "Eventos"),
    ("membresia", "Membresía"),
]


class MarkersMissing(Exception):
    pass


def _by_track(badges: list[Badge]) -> list[tuple[str, list[Badge]]]:
    known = {slug for slug, _ in TRACKS}
    groups = []
    for slug, title in TRACKS:
        items = sorted((b for b in badges if b.track == slug), key=lambda b: b.slug)
        if items:
            groups.append((title, items))
    others = sorted((b for b in badges if b.track not in known), key=lambda b: b.slug)
    if others:
        groups.append(("Otros", others))
    return groups


def _estado(b: Badge) -> str:
    if b.status == "ok":
        return "✅"
    if b.status == "regen":
        return "🔄 regenerar"
    return "❌ pendiente"


def render_readme_tables(m: Manifest) -> str:
    total = len(m.badges)
    sin_imagen = sum(1 for b in m.badges if b.status == "missing")
    parts = [
        "## Catálogo de badges",
        "",
        f"_Generado por `badgegen docs` desde `badges.yaml` ({total} badges activos, "
        f"{sin_imagen} sin imagen)._",
    ]
    for title, items in _by_track(m.badges):
        parts += ["", f"### {title}", "", "| Slug | Nombre | Imagen | Estado |",
                  "|------|--------|--------|--------|"]
        parts += [f"| `{b.slug}` | {b.name} | `{b.file}` | {_estado(b)} |" for b in items]
    return "\n".join(parts)


def render_missing(m: Manifest) -> str:
    pending = [b for b in m.badges if b.status in ("missing", "regen")]
    parts = [
        f"Badges sin imagen o marcados para regenerar ({len(pending)}). "
        "Convención: `badge-<slug>.png`.",
        "",
        "Tras subir una imagen, purga la caché de jsDelivr: "
        "`https://purge.jsdelivr.net/gh/makespacemadrid/open-badges@main/<filename>`.",
    ]
    for title, items in _by_track(pending):
        parts += ["", f"## {title}", "", "| Slug | Nombre | Archivo | Estado |",
                  "|------|--------|---------|--------|"]
        parts += [f"| `{b.slug}` | {b.name} | `{b.file}` | {_estado(b)} |" for b in items]
    return "\n".join(parts)


def replace_between(text: str, content: str) -> str:
    if START not in text or END not in text:
        raise MarkersMissing(
            f"faltan los marcadores {START} / {END}; añádelos alrededor de las tablas"
        )
    pattern = re.compile(re.escape(START) + ".*?" + re.escape(END), re.DOTALL)
    return pattern.sub(f"{START}\n{content}\n{END}", text)


def write_docs(settings: Settings, m: Manifest) -> None:
    for path, content in (
        (settings.readme_path, render_readme_tables(m)),
        (settings.missing_path, render_missing(m)),
    ):
        path.write_text(
            replace_between(path.read_text(encoding="utf-8"), content), encoding="utf-8"
        )
```

```python
"""Promoción de candidatos a imagen final y fix in-place de finales."""
from __future__ import annotations

import shutil
from pathlib import Path

import httpx
from PIL import Image

from badgegen import config, docs_gen, manifest as manifest_mod, postprocess
from badgegen.config import Settings
from badgegen.manifest import Manifest


def purge_cdn(file: str, client: httpx.Client | None = None) -> bool:
    client = client or httpx.Client()
    url = config.JSDELIVR_PURGE.format(file=file)
    try:
        response = client.get(url, timeout=10)
        if response.status_code < 400:
            return True
        print(f"WARNING: purge devolvió {response.status_code}; lánzalo a mano: {url}")
    except httpx.HTTPError as exc:
        print(f"WARNING: purge falló ({exc}); lánzalo a mano: {url}")
    return False


def _finish(settings: Settings, m: Manifest, badge, image: Image.Image,
            client: httpx.Client | None) -> Path:
    final = settings.repo_root / badge.file
    tmp = final.with_suffix(".tmp.png")
    image.save(tmp, format="PNG")
    tmp.replace(final)
    badge.status = "ok"
    badge.issues = []
    manifest_mod.save(m, settings.manifest_path)
    purge_cdn(badge.file, client=client)
    try:
        docs_gen.write_docs(settings, m)
    except docs_gen.MarkersMissing as exc:
        print(f"WARNING: docs no regenerados: {exc}")
    return final


def promote(settings: Settings, m: Manifest, slug: str, candidate: Path,
            keep: bool = False, client: httpx.Client | None = None) -> Path:
    badge = m.get(slug)
    if badge is None:
        raise KeyError(f"slug desconocido: {slug}")
    if not candidate.is_file():
        raise FileNotFoundError(candidate)
    processed = postprocess.process(
        Image.open(candidate), diameter=m.target_diameter
    )
    final = _finish(settings, m, badge, processed, client)
    if not keep:
        shutil.rmtree(settings.candidates_dir / slug, ignore_errors=True)
    return final


def fix(settings: Settings, m: Manifest, slug: str,
        client: httpx.Client | None = None) -> Path:
    badge = m.get(slug)
    if badge is None:
        raise KeyError(f"slug desconocido: {slug}")
    final = settings.repo_root / badge.file
    if not final.is_file():
        raise FileNotFoundError(final)
    processed = postprocess.process(Image.open(final), diameter=m.target_diameter)
    return _finish(settings, m, badge, processed, client)
```

```python
"""CLI de badgegen: sync | scan | generate | review | promote | fix | docs."""
from __future__ import annotations

import argparse
import json
import os
from pathlib import Path

from PIL import Image

from badgegen import catalog, docs_gen, gallery, generate as generate_mod, manifest as manifest_mod, promote as promote_mod, qc
from badgegen.config import Settings


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="badgegen",
                                     description="Generación y QC de imágenes de Open Badges")
    sub = parser.add_subparsers(dest="command", required=True)

    p_sync = sub.add_parser("sync", help="catálogo del portal → badges.yaml")
    p_sync.add_argument("--catalog", type=Path, default=None)

    p_scan = sub.add_parser("scan", help="QC de las imágenes existentes")
    p_scan.add_argument("slugs", nargs="*")
    p_scan.add_argument("--judge", action="store_true")
    p_scan.add_argument("--write", action="store_true")

    p_gen = sub.add_parser("generate", help="genera candidatos")
    p_gen.add_argument("slugs", nargs="*")
    p_gen.add_argument("--status", default="missing,regen")
    p_gen.add_argument("-n", type=int, default=4)
    p_gen.add_argument("--provider", choices=["gemini", "custom"], default=None)
    p_gen.add_argument("--model", default=None)

    p_rev = sub.add_parser("review", help="regenera la galería de candidatos")
    p_rev.add_argument("slugs", nargs="*")

    p_pro = sub.add_parser("promote", help="candidato → imagen final")
    p_pro.add_argument("slug")
    p_pro.add_argument("candidate", type=Path)
    p_pro.add_argument("--keep-candidates", action="store_true")

    p_fix = sub.add_parser("fix", help="post-proceso in-place del PNG final")
    p_fix.add_argument("slug")

    sub.add_parser("docs", help="regenera README.md y missing-images.md")
    return parser


def _load(settings: Settings) -> manifest_mod.Manifest:
    if settings.manifest_path.is_file():
        return manifest_mod.load(settings.manifest_path)
    return manifest_mod.Manifest()


def cmd_sync(settings: Settings, args) -> int:
    m = _load(settings)
    entries = catalog.parse_catalog(args.catalog or settings.catalog_path)
    warnings = catalog.merge_catalog(m, entries, settings.repo_root)
    manifest_mod.save(m, settings.manifest_path)
    counts = {s: sum(1 for b in m.badges if b.status == s) for s in ("ok", "missing", "regen")}
    print(f"{len(m.badges)} badges en manifiesto: {counts}")
    for warning in warnings:
        print(f"AVISO: {warning}")
    return 0


def cmd_scan(settings: Settings, args) -> int:
    m = _load(settings)
    badges = generate_mod.select_badges(m, args.slugs or None, None) if args.slugs \
        else [b for b in m.badges if b.status != "missing"]
    rows = []
    for badge in badges:
        path = settings.repo_root / badge.file
        if not path.is_file():
            continue
        result = qc.check_image(Image.open(path), threshold=m.roundness_threshold)
        judge_score, judge_issues = None, []
        if args.judge:
            verdict = qc.judge_legibility(path.read_bytes(), os.environ["GEMINI_API_KEY"])
            judge_score, judge_issues = verdict.score, verdict.issues
            if verdict.score <= 3:
                result.issues.append("low-legibility")
        rows.append({"slug": badge.slug, "file": badge.file,
                     "roundness": result.roundness, "issues": result.issues,
                     "judge_score": judge_score, "judge_issues": judge_issues})
        if args.write:
            badge.issues = list(result.issues)
        flag = "OK " if not result.issues else "!! "
        print(f"{flag}{badge.slug} redondez={result.roundness:.4f} {result.issues}")
    if args.write:
        manifest_mod.save(m, settings.manifest_path)
    out = gallery.write_scan_report(settings, rows)
    print(f"informe: {out}")
    return 0


def cmd_generate(settings: Settings, args) -> int:
    from badgegen.providers import get_provider

    m = _load(settings)
    provider = get_provider(args.provider, model=args.model)
    statuses = None if args.slugs else [s.strip() for s in args.status.split(",") if s.strip()]
    report = generate_mod.run_generate(settings, m, provider,
                                       slugs=args.slugs or None, statuses=statuses, n=args.n)
    errors = 0
    for slug, outcome in report.items():
        if isinstance(outcome, str):
            errors += 1
            print(f"!! {slug}: {outcome}")
        else:
            print(f"OK {slug}: {len(outcome)} candidatos")
    _write_review(settings, m, None)
    return 1 if errors else 0


def cmd_review(settings: Settings, args) -> int:
    m = _load(settings)
    _write_review(settings, m, args.slugs or None)
    return 0


def _write_review(settings: Settings, m, slugs) -> None:
    items = []
    if not settings.candidates_dir.is_dir():
        print("no hay candidatos")
        return
    for directory in sorted(settings.candidates_dir.iterdir()):
        if not directory.is_dir() or (slugs and directory.name not in slugs):
            continue
        badge = m.get(directory.name)
        candidates = []
        for png in sorted(directory.glob("*.png")):
            sidecar_path = png.with_suffix(".json")
            sidecar = json.loads(sidecar_path.read_text(encoding="utf-8")) if sidecar_path.is_file() else {}
            qc_data = sidecar.get("qc", {})
            candidates.append({
                "file": str(png.relative_to(settings.repo_root)),
                "qc_ok": not qc_data.get("issues"),
                "roundness": qc_data.get("roundness", 0.0),
                "provider": sidecar.get("provider", "?"),
                "model": sidecar.get("model", "?"),
            })
        if candidates:
            current = badge.file if badge and (settings.repo_root / badge.file).is_file() else None
            items.append({"slug": directory.name, "current": current, "candidates": candidates})
    out = gallery.write_review(settings, items)
    print(f"galería: {out}")


def cmd_promote(settings: Settings, args) -> int:
    m = _load(settings)
    candidate = args.candidate if args.candidate.is_absolute() \
        else settings.repo_root / args.candidate
    final = promote_mod.promote(settings, m, args.slug, candidate,
                                keep=args.keep_candidates)
    print(f"promocionado: {final}")
    return 0


def cmd_fix(settings: Settings, args) -> int:
    m = _load(settings)
    final = promote_mod.fix(settings, m, args.slug)
    print(f"corregido in-place: {final}")
    return 0


def cmd_docs(settings: Settings, args) -> int:
    m = _load(settings)
    docs_gen.write_docs(settings, m)
    print("README.md y missing-images.md regenerados")
    return 0


COMMANDS = {
    "sync": cmd_sync,
    "scan": cmd_scan,
    "generate": cmd_generate,
    "review": cmd_review,
    "promote": cmd_promote,
    "fix": cmd_fix,
    "docs": cmd_docs,
}


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    settings = Settings.locate()
    return COMMANDS[args.command](settings, args)
```
